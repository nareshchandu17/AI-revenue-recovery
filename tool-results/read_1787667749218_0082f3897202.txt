    43→class MockAIProvider implements AIProvider {
    44→  readonly name = "mock"
    45→  private response: AIResponse
    46→  private shouldFail: boolean = false
    47→  private failMessage: string = "Provider unavailable"
    48→
    49→  constructor(response: AIResponse) {
    50→    this.response = response
    51→  }
    52→
    53→  withFailure(msg: string = "Provider unavailable"): this {
    54→    this.shouldFail = true
    55→    this.failMessage = msg
    56→    return this
    57→  }
    58→
    59→  async complete(): Promise<AIResponse> {
    60→    if (this.shouldFail) {
    61→      throw new Error(this.failMessage)
    62→    }
    63→    return this.response
    64→  }
    65→}
